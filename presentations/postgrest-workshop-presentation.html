PostGIS Integration

-- Enable PostGIS
CREATE EXTENSION postgis;

-- Spatial data types
CREATE TABLE locations (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100),
    geom GEOMETRY(POINT, 4326)
);

-- Spatial queries
SELECT name FROM locations 
WHERE ST_DWithin(geom, ST_Point(-71.0589, 42.3601), 1000);

Result: Geographic APIs with zero additional code!

Database Schema → API

-- Create table
CREATE TABLE workshops (
    id SERIAL PRIMARY KEY,
    title VARCHAR(200) NOT NULL,
    duration INTEGER,
    instructor VARCHAR(100)
);

-- Grant permissions
GRANT SELECT ON workshops TO web_anon;

  • GET /workshops - List all
  • GET /workshops?id=eq.1 - Filter
  • GET /workshops?select=title,duration - Projection

Row Level Security

-- Enable RLS
ALTER TABLE workshops ENABLE ROW LEVEL SECURITY;

-- Create policy
CREATE POLICY workshop_policy ON workshops
  FOR ALL TO workshop_user
  USING (instructor = current_setting('request.jwt.claims')::json->>'email');

Result: Users only see their own workshops automatically!

Custom Functions

Business Logic

PostgreSQL Functions → API Endpoints

-- Function becomes RPC endpoint
CREATE FUNCTION search_workshops(query TEXT)
RETURNS TABLE(id INT, title TEXT, relevance REAL) AS $$
BEGIN
  RETURN QUERY
  SELECT w.id, w.title, 
         similarity(w.title, query) as relevance
  FROM workshops w
  WHERE w.title % query
  ORDER BY relevance DESC;
END;
$$ LANGUAGE plpgsql;

API call:

POST /rpc/search_workshops
{"query": "PostgREST"}

Spatial Functions

CREATE FUNCTION nearby_locations(
  lat FLOAT, 
  lng FLOAT, 
  radius_meters INT DEFAULT 1000
)
RETURNS TABLE(name TEXT, distance FLOAT) AS $$
BEGIN
  RETURN QUERY
  SELECT l.name, 
         ST_Distance(l.geom, ST_Point(lng, lat)::geography) as distance
  FROM locations l
  WHERE ST_DWithin(l.geom, ST_Point(lng, lat)::geography, radius_meters)
  ORDER BY distance;
END;
$$ LANGUAGE plpgsql;

When NOT to Use PostgREST

  • File processing - Heavy computational tasks
  • Third-party integrations - External API orchestration
  • Custom protocols - Non-HTTP requirements
  • Legacy constraints - Existing architecture limitations

Best Practices

  • Schema design first - API follows structure
  • Use views for complex queries
  • Implement RLS for security
  • Create indexes for performance
  • Version your schema with migrations
  • Monitor query performance regularly
  • Test with realistic data volumes
